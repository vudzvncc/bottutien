function normalize(s){return(s||'').toLowerCase().replace(/\s+/g,' ').trim()}

function cleanupKeyword(kw){
    if(!kw) return '';
    let s = String(kw).toLowerCase();
    s = s.replace(/https?:\/\/\S+/g, ' ');
    s = s.replace(/\b[a-z0-9-]+(?:\.[a-z0-9-]+){1,4}\b/gi, ' ');
    s = s.replace(/[^\p{L}\p{N}\s]/gu, ' ').replace(/\s+/g, ' ').trim();
    return s;
}

function lookupEntries(kw){
    const m=loadMap();
    const k=normalize(kw);
    const kClean = normalize(cleanupKeyword(kw));
    for(const[kw2,entries]of Object.entries(m)){
        const kw2n = normalize(kw2);
        if(kw2n===k || kw2n===kClean) return entries.slice();
    }
    for(const[kw2,entries]of Object.entries(m)){
        const kw2n = normalize(kw2);
        if(kClean && (kClean.includes(kw2n) || kw2n.includes(kClean))) return entries.slice();
    }
    for(const[kw2,entries]of Object.entries(m)){
        const kw2n = normalize(kw2);
        if(k.includes(kw2n)) return entries.slice();
    }
    return [];
}

function cleanHintString(str) {
    if (!str) return '';
    return String(str)
        .normalize('NFC')
        .toLowerCase()
        .replace(/[^\p{L}\p{N}]/gu, '');
}

function levenshtein(a, b) {
    const m = a.length;
    const n = b.length;
    if (m === 0) return n;
    if (n === 0) return m;
    const dp = Array.from({ length: m + 1 }, () => new Array(n + 1).fill(0));
    for (let i = 0; i <= m; i++) dp[i][0] = i;
    for (let j = 0; j <= n; j++) dp[0][j] = j;
    for (let i = 1; i <= m; i++) {
        for (let j = 1; j <= n; j++) {
            const cost = a[i - 1] === b[j - 1] ? 0 : 1;
            dp[i][j] = Math.min(
                dp[i - 1][j] + 1,
                dp[i][j - 1] + 1,
                dp[i - 1][j - 1] + cost
            );
        }
    }
    return dp[m][n];
}

function getSimilarity(s1, s2) {
    const dist = levenshtein(s1, s2);
    const maxLen = Math.max(s1.length, s2.length);
    if (maxLen === 0) return 1.0;
    return 1.0 - (dist / maxLen);
}

function exactHintMatch(mapEntry, hintTexts){
    if(!mapEntry || !mapEntry.hint) return false;
    const mapClean = cleanHintString(mapEntry.hint);
    if (mapClean.length < 5) return false;
    const ocrJoined = hintTexts.join(' ');
    const ocrClean = cleanHintString(ocrJoined);
    if (ocrClean.length < 5) return false;

    if (ocrClean.includes(mapClean) || mapClean.includes(ocrClean)) {
        console.log(`[HINT MATCH] ✓ Substring 100%`);
        return true;
    }

    let bestScore = 0;
    const mapLen = mapClean.length;
    const windowSizes = [mapLen, Math.floor(mapLen * 0.9), Math.floor(mapLen * 1.1)];

    for (const winSize of windowSizes) {
        if (winSize <= 0 || winSize > ocrClean.length) continue;
        for (let i = 0; i <= ocrClean.length - winSize; i++) {
            const sub = ocrClean.substring(i, i + winSize);
            const score = getSimilarity(mapClean, sub);
            if (score > bestScore) bestScore = score;
            if (score >= 0.90) {
                console.log(`[HINT MATCH] ✓ Fuzzy 90% (Score: ${(score*100).toFixed(1)}%)`);
                return true;
            }
        }
    }
    console.log(`[HINT MATCH] ✗ Tạch. Best Score: ${(bestScore*100).toFixed(1)}% | Map: "${mapClean}" | OCR: "${ocrClean}"`);
    return false;
}

    }

    const matched = entries.filter(e => exactHintMatch(e, texts));

    if(!matched.length){
        log(`⚠ pickEntry: HINT OCR KHÔNG KHỚP BẤT KỲ entry nào trong map. DỪNG.`,'warn');
        return {__unresolved:true, reason:'no-match', entries, hintTexts:texts};
    }

    if(matched.length > 1){
        log(`⚠ pickEntry: Có ${matched.length} entry cùng khớp hint → chọn đại entry đầu tiên.`,'warn');
        return matched[0];
    }

    log(`✓ pickEntry EXACT: "${matched[0].hint}" → ${matched[0].domain}`);
    return matched[0];
}

function isValidCode(s){
    if(!s)return false;
    const t=String(s).trim();
    if(!CONFIG.codeRegex.test(t))return false;
    if(CONFIG.codeBlacklist.test(t))return false;
    const hu=/[A-Z]/.test(t);
    const hl=/[a-z]/.test(t);
    const hd=/\d/.test(t);
    return (hu && hl) || (hu && hd) || (hl && hd);
}

function extractCodeFromText(raw){
    if(!raw)return null;
    const s=String(raw).trim();
    if(isValidCode(s))return s;
    const matches=s.match(/\b[A-Za-z0-9]{6,12}\b/g);
    if(matches){
        for(const cand of matches){
            if(isValidCode(cand))return cand;
        }
    }
    return null;
}

function scanServerPayloadForCode(payload){
    if(!payload) return null;
    if(typeof payload === 'string'){
        try{
            const parsed = JSON.parse(payload);
            const fromJson = scanServerPayloadForCode(parsed);
            if(fromJson) return fromJson;
        }catch(_){}
        const keyMatch = payload.match(/["']?(?:code|otp|token|verify_code|confirm_code|key|result|data)["']?\s*[:=]\s*["']([A-Za-z0-9]{6,12})["']/i);
        if(keyMatch && keyMatch[1] && isValidCode(keyMatch[1])) return keyMatch[1];
        return null;
    }
    if(typeof payload === 'object'){
        const priorityKeys = ['code','otp','token','verifyCode','verify_code','confirmCode','confirm_code','key','value'];
        for(const k of priorityKeys){
            if(Object.prototype.hasOwnProperty.call(payload, k)){
                const val = payload[k];
                if(typeof val === 'string' && isValidCode(val.trim())) return val.trim();
            }
        }
        for(const v of Object.values(payload)){
            if(v && typeof v === 'object'){
                const nested = scanServerPayloadForCode(v);
                if(nested) return nested;
            }
        }
    }
    return null;
}

function applyStrictColorMask(ctx,w,h,minH){
    const imgData=ctx.getImageData(0,0,w,h);
    const d=imgData.data;
    const W=w,H=h;
    const mask=new Uint8Array(W*H);
    const bl=CONFIG.colorMask.blueLink;
    const pm=CONFIG.colorMask.purple;

    for(let y=0;y<H;y++)for(let x=0;x<W;x++){
        const i=(y*W+x)*4;
        const r=d[i],g=d[i+1],b=d[i+2];

        const isBlue = (b > r + 30) && (b > g + 10) && (b > 100);
        const isPurple = (r > 60 && b > 80) && (b > g + 30) && (r > g + 10);

        if(isBlue || isPurple) mask[y*W+x]=1;
    }

    const visited=new Uint8Array(W*H);
    const stack=[];

    for(let y=0;y<H;y++)for(let x=0;x<W;x++){
        if(!mask[y*W+x]||visited[y*W+x])continue;
        stack.length=0;
        stack.push([x,y]);
        let minY=y,maxY=y;
        while(stack.length){
            const[cx,cy]=stack.pop();
            if(cx<0||cx>=W||cy<0||cy>=H)continue;
            if(visited[cy*W+cx])continue;
            if(!mask[cy*W+cx])continue;
            visited[cy*W+cx]=1;
            if(cy<minY)minY=cy;
            if(cy>maxY)maxY=cy;
            stack.push([cx+1,cy],[cx-1,cy],[cx,cy+1],[cx,cy-1]);
        }
        const hgt=maxY-minY+1;
        if(hgt<minH){
            stack.length=0;
            stack.push([x,y]);
            while(stack.length){
                const[cx,cy]=stack.pop();
                if(cx<0||cx>=W||cy<0||cy>=H)continue;
                if(!mask[cy*W+cx])continue;
                mask[cy*W+cx]=0;
                stack.push([cx+1,cy],[cx-1,cy],[cx,cy+1],[cx,cy-1]);
            }
        }
    }

    for(let y=0;y<H;y++)for(let x=0;x<W;x++){
        const i=(y*W+x)*4;
        const v=mask[y*W+x]?0:255;
        d[i]=d[i+1]=d[i+2]=v;
        d[i+3]=255;
    }
    ctx.putImageData(imgData,0,0);
}

function scanCodeInMainBtn(){
    if(gtfServerCodeCache && isValidCode(gtfServerCodeCache)) return gtfServerCodeCache;

    const sels=[
        '.trade-btn-clfs1__content',
        '.trade-btn-clfs1-content',
        '.gtr2-trade-btn__content',
        '.gtr2-copy-allowed',
        '.copy-allowed',
        '.gtr2-trade-btn',
        '#gtr2-trade-btn',
        '#trade-btn-clfs1',
        '.trade-btn-clfs1',
        '.trade-btn-clf__content',
        '.trade-btn-clf'
    ];
    for(const sel of sels){
        const nodes=document.querySelectorAll(sel);
        for(const el of nodes){
            if(!el)continue;
            const direct=extractCodeFromText(el.innerText || el.textContent || '');
            if(direct)return direct;

            const w=document.createTreeWalker(el,NodeFilter.SHOW_TEXT,null);
            let n;
            while((n=w.nextNode())){
                const found=extractCodeFromText(n.nodeValue||'');
                if(found)return found;
            }

            const title=el.getAttribute('title') || el.getAttribute('data-code') || el.getAttribute('data-copy');
            const fromAttr=extractCodeFromText(title||'');
            if(fromAttr)return fromAttr;
        }
    }

    const codeInputs = document.querySelectorAll('input[data-code], [data-code]');
    for(const item of codeInputs){
        const dc = extractCodeFromText(item.getAttribute('data-code') || item.value || '');
        if(dc) return dc;
    }

    return null;
}

async function scanCodeWithFallbackOcr(){
    const domCode=scanCodeInMainBtn();
    if(domCode)return domCode;

    const btn=document.querySelector('.trade-btn-clfs1__content, .gtr2-trade-btn__content, .gtr2-trade-btn, .trade-btn-clfs1, .trade-btn-clf__content, .trade-btn-clf');
    if(!btn)return null;
    const rect=btn.getBoundingClientRect();
    if(rect.width < 12 || rect.height < 8)return null;

    const txt=(btn.innerText || btn.textContent || '').replace(/\s+/g,'').trim();
    if(txt.length >= 6 && txt.length <= 14 && !/nhận|mã|xác|click|lấy|đổi|từ|khóa/i.test(txt)){
        const cleaned=txt.replace(/[^A-Za-z0-9]/g,'');
        if(isValidCode(cleaned))return cleaned;
    }

    try{
        if(!tesseractWorker && !tesseractWorkerPromise)return null;
        const w=await getWorker();
        if(typeof window.html2canvas==='function'){
            const snap=await window.html2canvas(btn,{backgroundColor:'#ffffff',scale:3});
            const res=await withTimeout(w.recognize(snap.toDataURL('image/png')),8000,'ocr-btn');
            return extractCodeFromText(res.data.text);
        }
    }catch(_){}
    return null;
}

async function waitForCodeMainBtn(timeout=180000){
    const s=Date.now();
    while(Date.now()-s<timeout){
        const c=await scanCodeWithFallbackOcr();
        if(c)return c;
        await sleep(SPEED.codePollInterval);
    }
    return null;
}

async function resolveEntryByHint(entries, kwText){
    if(!entries || !entries.length) return {ok:false, reason:'no-entries'};
    const hintImg = await waitFor(CONFIG.selectors.googleHintImage, 8000);
    if(!hintImg){
        logPanel(`⚠ Không thấy ảnh hint`, 'warn');
        return {ok:false, reason:'no-hint-img'};
    }
    let lastResult = null;
    for(let attempt=1; attempt<=CONFIG.hintRetryMax; attempt++){
        uiStatus(`OCR hint lần ${attempt}/${CONFIG.hintRetryMax}...`, 'active');
        let colored='', raw='';
        try{
            await ensureImageLoaded(hintImg);
            const r = await ocrGoogleHintImage(hintImg);
            colored = r.colored || '';
            raw = r.raw || '';
        }catch(e){
            logPanel(`OCR hint lần ${attempt} throw: ${e.message}`,'warn');
        }
        const texts = [colored, raw].filter(Boolean);
        logPanel(`Lần ${attempt}: hint colored="${colored.substring(0,60)}" | raw="${raw.substring(0,60)}"`, texts.length?'info':'warn');
        if(!texts.length){
            lastResult = {ok:false, reason:'ocr-empty', attempt};
            await sleep(CONFIG.hintRetryDelayMs * attempt);
            continue;
        }
        State.set('hintText', colored);
        State.set('hintRawText', raw);
        uiHint(colored || raw);

        const pick = pickEntry(entries, texts);
        if(pick && !pick.__unresolved){
            logPanel(`✓ Campaign resolved: ${pick.domain} (lần ${attempt})`,'ok');
            State.set('chosenHint', pick.hint);
            return {ok:true, entry:pick, attempt};
        }
        lastResult = {ok:false, reason:pick?.reason || 'no-match', attempt, pick, texts};
        logPanel(`✗ Lần ${attempt} chưa resolve (${lastResult.reason})`,'warn');
        if(attempt < CONFIG.hintRetryMax){
            await sleep(CONFIG.hintRetryDelayMs * attempt);
        }
    }
    return {ok:false, ...lastResult, retriesExhausted:true};
}

async function handleGTrafficAsProducer(){
    const originUrl = location.href;
    const phase = State.get('phase');
    logPanel(`GTraffic producer, phase: ${phase}`,'info');

    if(phase === PHASE.GOOGLE_SEARCH){
        const hasCode = State.get('capturedCode') || GM_getValue('gtf_captured_code', null);
        const tabOpen = State.get('workTabOpened');
        if(!hasCode && tabOpen !== true){
            logPanel('⚠ Phase google_search stale → reset','warn');
            State.clear();
            State.set('phase', PHASE.IDLE);
            uiStatus('Sẵn sàng','');
            return;
        }
    }

    const targetFailed = GM_getValue('targetSiteFailed', null);
    if(targetFailed){
        logPanel(`⚠ Target site lỗi: ${targetFailed}`,'warn');
        GM_setValue('targetSiteFailed', null);
        if(phase===PHASE.GOOGLE_SEARCH || phase===PHASE.TARGET_SITE || phase===PHASE.RETURN_GTRAFFIC){
            logPanel(`Đổi từ khóa do target lỗi (chờ ${CONFIG.targetFailedTimeout/1000}s)...`,'info');
            uiStatus(`Đổi từ khóa mới...`,'warn');
            await sleep(CONFIG.targetFailedTimeout);
            const clicked = await clickRefreshKeyword();
            if(clicked){
                State.clear();
                GM_setValue('firstClickDone', null);
                await sleep(2000);
                logPanel('Đã đổi từ khóa','ok');
                if(autoMode){
                    logPanel('🤖 AUTO — chạy lại OCR','ok');
                    const domain = await runStartOCR();
                    if(!domain) return;
                    State.set('originUrl', location.href);
                    State.set('phase', PHASE.GOOGLE_SEARCH);
                    State.set('capturedCode', null);
                    const keyword = State.get('keyword') || domain;
                    logPanel(`Mở tab work → Google: "${keyword}"`,'ok');
                    await midDelay();
                    const tab = window.open(`https://www.google.com/search?q=${encodeURIComponent(keyword)}`, '_blank');
                    if(tab){ State.set('workTabOpened', true); uiStatus('Chờ code từ tab work...','active'); waitForCodeAndFill(); }
                    else{
                        logPanel('Popup bị chặn — tự chuyển trang hiện tại sang Google','warn');
                        State.set('workTabOpened', true);
                        uiStatus('Chờ code từ tab work...','active');
                        location.href = `https://www.google.com/search?q=${encodeURIComponent(keyword)}`;
                    }
                } else {
                    uiStatus('Đã đổi từ khóa — bấm BẮT ĐẦU','warn');
                }
                return;
            }
        }
    }

    if(!phase || phase===PHASE.IDLE){
        if(autoMode){
            logPanel('🤖 AUTO','ok');
            const domain = await runStartOCR();
            if(!domain) return;
            State.set('originUrl', originUrl);
            State.set('phase', PHASE.GOOGLE_SEARCH);
            State.set('capturedCode', null);
            const keyword = State.get('keyword') || domain;

            const searchUrl = `https://www.google.com/search?q=${encodeURIComponent(keyword)}`;
            logPanel(`Mở tab work → Google tìm Keyword: "${keyword}"`,'ok');

            try {
                GM_setValue('gtf_phase', PHASE.TARGET_SITE);
                GM_setValue('gtf_targetDomain', domain);
                GM_setValue('gtf_keyword', keyword);
            } catch(_){}

            await midDelay();

            try{
                const tab = window.open(searchUrl, '_blank');
                if(!tab){
                    logPanel('Popup bị chặn — tự chuyển trang hiện tại sang Google','warn');
                    location.href = searchUrl;
                    return;
                }
            }catch(e){
                logPanel(`window.open throw: ${e.message}`,'err');
                alert('Không mở được tab Google — kiểm tra popup blocker');
                return;
            }
            State.set('workTabOpened', true);
            uiStatus('Chờ người dùng click...','active');
            waitForCodeAndFill();
            return;
        }
        return;
    }
    if(phase===PHASE.GOOGLE_SEARCH){ uiStatus('Chờ code từ tab work...','active'); waitForCodeAndFill(); return; }
    if(phase===PHASE.GET_LINK){ await autoClickGetLink(); return; }
}

async function waitForCodeAndFill(){

    const start=Date.now();let lastLog=0;
    while(Date.now()-start<SPEED.codePollMaxWait){
        const targetFailed=GM_getValue('targetSiteFailed',null);
        if(targetFailed){
            logPanel(`⚠ Target lỗi: ${targetFailed}`,'warn');
            GM_setValue('targetSiteFailed',null);
            State.clear();

            await clickRefreshKeyword();

        }

        const codeFromChild = GM_getValue('gtf_captured_code', null);
        if (codeFromChild && isValidCode(codeFromChild)) {

function unwrapGoogleRedirect(href){
    try{
        if(!href) return href;
        if(!/google\./i.test(href)) return href;
        const u = new URL(href, location.origin);
        const q = u.searchParams.get('q') || u.searchParams.get('url');
        if(q && /^https?:\/\//i.test(q)) return q;
    }catch(_){}
    return href;
}

async function handleGoogleConsumer(){
    if(State.get('phase') !== PHASE.GOOGLE_SEARCH) return;

    const targetDomain = extractHost(State.get('targetDomain') || GM_getValue('gtf_targetDomain') || '');
    const keyword = (State.get('keyword') || '').trim();
    if(!targetDomain) return;

    logPanel(`Google: keyword "${keyword || '(none)'}", target: "${targetDomain}"`,'info');
    uiStatus('Đang tìm link Domain...','active');

    let isMoved = false;
    let pageCount = 1;
    let attemptsNoResult = 0;
    const maxNoResultAttempts = 20;

    const finder = setInterval(() => {
        if (isMoved) return;
        
        const links = document.querySelectorAll('a[href]');
        for(const a of links){
            try{
                const rawHref = a.href || '';
                const realHref = unwrapGoogleRedirect(rawHref);
                if(!/^https?:\/\//i.test(realHref)) continue;
                const u = new URL(realHref);
                
                const host = u.hostname.replace(/^www\./,'').toLowerCase();
                if(host === targetDomain || host.includes(targetDomain)){
                    isMoved = true;
                    clearInterval(finder);
                    logPanel(`Đã tìm thấy link ${realHref}`, 'ok');
                    window.open(realHref, '_blank');
                    try{ window.close(); }catch(_){}
                    return;
                }
            }catch(_){}
        }

        attemptsNoResult++;
        
        if(attemptsNoResult > maxNoResultAttempts) {
            const nextBtn = document.querySelector('#pnnext, a[aria-label="Next"], a[aria-label="Trang tiếp theo"]');
            if (nextBtn && pageCount < 3) {
                logPanel(`Không thấy Domain. Chuyển sang trang Google ${pageCount + 1}...`, 'warn');
                pageCount++;
                attemptsNoResult = 0;
                try { nextBtn.click(); } catch(e){}
            } else if (pageCount >= 3) {
                isMoved = true;
                clearInterval(finder);
                logPanel(`❌ Không tìm thấy trên Google. Mở trực tiếp: https://${targetDomain}`, 'warn');
                window.open(`https://${targetDomain}`, '_blank');
                try{ window.close(); }catch(_){}
            }
        }
    }, 250);
}

function getBrowserZoomScale(){
    try{
        if(window.visualViewport && typeof window.visualViewport.scale === 'number' && window.visualViewport.scale > 0){
            return Number(window.visualViewport.scale.toFixed(2));
        }
    }catch(_){}
    return 1;
}

function getViewportCenterCoords(){
    try{
        if(window.visualViewport){
            return {
                x: window.visualViewport.offsetLeft + window.visualViewport.width / 2,
                y: window.visualViewport.offsetTop + window.visualViewport.height / 2
            };
        }
    }catch(_){}
    return {
        x: window.innerWidth / 2,
        y: window.innerHeight / 2
    };
}

function isGreenishElement(el){
    if(!el) return false;
    try{
        const st = window.getComputedStyle(el);
        const bg = st.backgroundColor || '';
        const color = st.color || '';
        const border = st.borderColor || '';
        const combined = `${bg} ${color} ${border}`;
        const matches = [...combined.matchAll(/rgba?\(\s*([\d.]+)\s*,\s*([\d.]+)\s*,\s*([\d.]+)/gi)];
        for(const m of matches){
            const r = parseFloat(m[1]);
            const g = parseFloat(m[2]);
            const b = parseFloat(m[3]);
            if(g >= 115 && g > r + 20 && g > b + 15) return true;
        }
        const cls = String(el.className || '').toLowerCase();
        const id = String(el.id || '').toLowerCase();
        if(/green|success|emerald|lime|gtr2|trade-btn|clf/.test(`${cls} ${id}`)) return true;
    }catch(_){}
    return false;
}

function resolveAnchorElement(el){
    if(!el || !el.parentElement) return el;
    const p = el.parentElement;
    if(p === document.body || p === document.documentElement) return el;
    try{
        const rEl = el.getBoundingClientRect();
        const rP = p.getBoundingClientRect();
        const stP = window.getComputedStyle(p);
        const sameRow = Math.abs(rP.top - rEl.top) <= 18 && Math.abs(rP.bottom - rEl.bottom) <= 18;
        const reasonableBox = rP.width > rEl.width && rP.width <= 260 && rP.height >= rEl.height && rP.height <= 140;
        if(sameRow && reasonableBox && stP.display !== 'none' && stP.visibility !== 'hidden'){
            return p;
        }
    }catch(_){}
    return el;
}

function locateGreenGButton(){
    const selectors = [
        '#gtr2-trade-btn',
        '#trade-btn-clfs1',
        '.gtr2-trade-btn',
        '.trade-btn-clfs1',
        '.trade-btn-clf',
        '.trade-btn-clfs1-container',
        '.gtr2-trade-btn__content',
        '.trade-btn-clfs1__content',
        '.trade-btn-clfs1-content',
        '.trade-btn-clf__content',
        '.gtr2-copy-allowed'
    ];

    for(const sel of selectors){
        const nodes = document.querySelectorAll(sel);
        for(const el of nodes){
            if(!el || !el.isConnected) continue;
            const st = window.getComputedStyle(el);
            const r = el.getBoundingClientRect();
            if(st.display !== 'none' && st.visibility !== 'hidden' && r.width > 4 && r.height > 4){
                return resolveAnchorElement(el);
            }
        }
    }

    const all = document.querySelectorAll('button, a, div, span');
    let best = null;
    let bestScore = -1;

    for(const el of all){
        if(!el || !el.isConnected) continue;
        const r = el.getBoundingClientRect();
        if(r.width < 10 || r.width > 180 || r.height < 10 || r.height > 180) continue;
        const txt = (el.textContent || '').trim();
        const isG = (txt === 'g' || txt === 'G');
        const hasGChild = !!el.querySelector && Array.from(el.querySelectorAll('span, div, b, strong')).some(c => {
            const ct = (c.textContent || '').trim();
            return ct === 'g' || ct === 'G';
        });
        if(!isG && !hasGChild) continue;

        const st = window.getComputedStyle(el);
        if(st.display === 'none' || st.visibility === 'hidden' || parseFloat(st.opacity || '1') <= 0.05) continue;

        let score = 0;
        if(isG) score += 2;
        if(hasGChild) score += 1;
        if(isGreenishElement(el) || (el.parentElement && isGreenishElement(el.parentElement))) score += 5;

        if(score > bestScore){
            bestScore = score;
            best = el;
        }
    }

    return resolveAnchorElement(best);
}

function getScrollableContainer(el){
    let cur = el ? el.parentElement : null;
    while(cur && cur !== document.documentElement && cur !== document.body){
        const st = window.getComputedStyle(cur);
        const oy = st.overflowY || '';
        const ox = st.overflowX || '';
        if((oy === 'auto' || oy === 'scroll' || oy === 'overlay') && cur.scrollHeight > cur.clientHeight + 4){
            return cur;
        }
        if((ox === 'auto' || ox === 'scroll' || ox === 'overlay') && cur.scrollWidth > cur.clientWidth + 4){
            return cur;
        }
        cur = cur.parentElement;
    }
    return null;
}

function centerElementInViewportOnce(el){
    if(!el || !el.isConnected) return { found: true, centered: false, coords: null };
    const rect = el.getBoundingClientRect();
    if(rect.width < 1 || rect.height < 1) return { found: true, centered: false, coords: null };

    const vpCenter = getViewportCenterCoords();
    const elCenterX = rect.left + rect.width / 2;
    const elCenterY = rect.top + rect.height / 2;

    const deltaX = elCenterX - vpCenter.x;
    const deltaY = elCenterY - vpCenter.y;
    const tol = CONFIG.centerTolerancePx || 15;

    if(Math.abs(deltaX) > tol || Math.abs(deltaY) > tol){
        const subScroll = getScrollableContainer(el);
        if(subScroll){
            try{ subScroll.scrollTop += deltaY; }catch(_){}
            try{ subScroll.scrollLeft += deltaX; }catch(_){}
        }
        try{
            window.scrollBy({
                left: deltaX,
                top: deltaY,
                behavior: 'instant'
            });
        }catch(_){
            try{
                window.scrollTo(window.scrollX + deltaX, window.scrollY + deltaY);
            }catch(__){}
        }
    }

    const r2 = el.getBoundingClientRect();
    const vpCenter2 = getViewportCenterCoords();
    const cx2 = Math.round(r2.left + r2.width / 2);
    const cy2 = Math.round(r2.top + r2.height / 2);
    const centered = Math.abs(cx2 - Math.round(vpCenter2.x)) <= tol && Math.abs(cy2 - Math.round(vpCenter2.y)) <= tol;

    return {
        found: true,
        centered,
        coords: { x: cx2, y: cy2, width: Math.round(r2.width), height: Math.round(r2.height) }
    };
}

        btnT.style.outline = mode === 'trafficso1' ? '2px solid #facc15' : 'none';
    }
    if(status && statusText){
        status.innerText = statusText;
        if(statusColor) status.style.color = statusColor;
    } else if(status && !statusText){
        if(needIp && mode){
            const label = mode === 'gtraffic' ? 'GTraffic' : 'TrafficSO1';
            status.innerText = `Đã xong 2 link ${label} — HÃY ĐỔI IP!`;
            status.style.color = '#f87171';
        } else if(mode){
            const label = mode === 'gtraffic' ? 'GTraffic' : 'TrafficSO1';
            status.innerText = `Chế độ AUTO: ${label} (${counts[mode]}/2)`;
            status.style.color = '#4ade80';
        } else {
            status.innerText = 'Sẵn sàng';
            status.style.color = '#94a3b8';
        }
    }
}

async function runTaskFlow(type){
    if(ttnBusy) return;
    ttnBusy = true;
    try{
        const status = document.querySelector('#ttn-panel #ttn-status');
        const label = type === 'gtraffic' ? 'GTraffic' : 'TrafficSO1';
        updateTaskTrafficPanelUI(`Đang tìm nút tạo link ${label}...`, '#facc15');

        const startBtn = findTaskCardButton(type);
        if(!startBtn){
            if(status){
                status.innerText = `Chưa thấy nút ${label} — đang chờ/đổi IP...`;
                status.style.color = '#facc15';
            }
            return;
        }

        updateTaskTrafficPanelUI(`Đang click Bắt đầu vượt link ${label}...`, '#facc15');
        await HumanSim.click(startBtn);

        updateTaskTrafficPanelUI(`Đang chờ tạo link & hiện "Mở Tại Trang Này"...`, '#facc15');
        const deadline = Date.now() + 20000;
        let openBtn = null;
        while(Date.now() < deadline){
            openBtn = findModalOpenHereBtn();
            if(openBtn) break;
            await sleep(250);
        }

        if(!openBtn){
            updateTaskTrafficPanelUI(`Không thấy nút "Mở Tại Trang Này"!`, '#ef4444');
            return;
        }

        const targetHref = openBtn.getAttribute('href') || '';
        updateTaskTrafficPanelUI(`Đang mở trang đối tác: ${targetHref.substring(0, 32)}...`, '#4ade80');
        await HumanSim.click(openBtn);
        await sleep(400);
        if(targetHref && /^https?:\/\//i.test(targetHref) && location.href !== targetHref){
            location.href = targetHref;
        }
    }finally{
        ttnBusy = false;
    }
}

    });
    btnT.addEventListener('click', () => {
        setDoneCounts({ gtraffic: 0, trafficso1: 0 });

let ttnIpCheckRunning = false;

async function handleTaskTrafficNgon(){
    recordCompletedTaskIfNeeded();
    buildTaskTrafficPanel();

    if(!isNhiemVuPage()){
        startBackToNhiemVuWatcher();
        const backBtn = findBackToNhiemVuBtn();
        if(backBtn && !ttnBackClickDone) {
            ttnBackClickDone = true;
            if(ttnBackPollTimer){ clearInterval(ttnBackPollTimer); ttnBackPollTimer = null; }
            logPanel('Tự động click "Quay về trang nhiệm vụ"', 'ok');
            await sleep(200);
            await triggerBackToNhiemVu(backBtn);
        }
        return;
    }

    const stickyMode = getStickyTaskMode();
    const counts = getDoneCounts();
    if(stickyMode && (isNeedChangeIp() || (counts[stickyMode] || 0) >= 2)){
        setNeedChangeIp(true);
        const label = stickyMode === 'gtraffic' ? 'GTraffic' : 'TrafficSO1';
        updateTaskTrafficPanelUI(`Đã vượt xong 2 link ${label} — Yêu cầu người dùng ĐỔI IP!`, '#f87171');
        logPanel(`Đã vượt xong 2 link ${label}. Yêu cầu người dùng ĐỔI IP rồi bấm lại nút chế độ.`, 'warn');
        return;
    }

    if(stickyMode && !ttnBusy && !ttnIpCheckRunning){
        const startBtn = findTaskCardButton(stickyMode);
        if(startBtn){
            await sleep(700);
            if(isNhiemVuPage() && !ttnBusy && !ttnIpCheckRunning && !isNeedChangeIp()){
                await runTaskFlow(stickyMode);
            }
        }
    }

    if(ttnIpCheckRunning) return;
    const hasIpLimitTag = !!document.querySelector('.tag-danger');
    const checkBtn = document.querySelector('a[data-check-ip]');
    if(hasIpLimitTag && checkBtn && hasSize(checkBtn)) {
        ttnIpCheckRunning = true;
        logPanel('Bị giới hạn IP. Yêu cầu đổi IP. Đợi 45s...', 'warn');
        let timeLeft = 45;
        const interval = setInterval(() => {
            timeLeft--;
            logPanel(`Đợi đổi IP... ${timeLeft}s`, 'warn');
            if(timeLeft <= 0) {
                clearInterval(interval);
                ttnIpCheckRunning = false;
                logPanel('Hết 45s. Tự động click Kiểm tra lại!', 'ok');
                HumanSim.click(checkBtn);
            }
        }, 1000);
    }
}

                    domain = await runStartOCR();

                    logPanel(`runStartOCR throw: ${e.message}`,'err');
                    domain = null;
                }

                if(!domain){
                    logPanel('OCR/hint không resolve được → hủy mở Google','warn');

                    GM_setValue('gtf_phase', PHASE.TARGET_SITE);
                    GM_setValue('gtf_targetDomain', domain);

                } catch(_){}

                await midDelay();
